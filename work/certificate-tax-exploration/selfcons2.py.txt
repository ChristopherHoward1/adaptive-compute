"""Exp 3: adaptive self-consistency on real repeated-sampling data (monkey_business).
Each question's ~10k samples define its answer distribution pi_q; policies draw iid from pi_q.
Scored on accuracy vs ground truth and agreement with the k->inf mode. R reps per question."""
import pickle, sys
import numpy as np
from scipy.special import betaln, betainc
from scipy.stats import beta as Beta

rng = np.random.default_rng(int(sys.argv[2]) if len(sys.argv) > 2 else 7)
TOP, R, MMAX = 30, 300, 256
qs = pickle.load(open(sys.argv[1], 'rb'))

def prep(q):
    items = sorted(q['answers'].items(), key=lambda kv: -kv[1])
    items = [(a, c) for a, c in items if a is not None] + [(a, c) for a, c in items if a is None]
    n = sum(c for _, c in items)
    top = items[:TOP]
    probs = np.array([c for _, c in top] + [n - sum(c for _, c in top)], float) / n
    corr = np.array([a in q['correct'] and a is not None for a, _ in top])
    # None ("no answer extracted") never counts as a vote
    valid = np.array([a is not None for a, _ in top])
    return probs, corr, valid

def runs(probs, valid):
    k = len(probs) - 1
    draws = rng.choice(k+1, size=(R, MMAX), p=probs)       # index k == tail (unique answers)
    ok = (draws < k)
    ok[ok] = valid[draws[ok]]
    onehot = np.zeros((R, MMAX, k), np.int16)
    r, t = np.nonzero(ok)
    onehot[r, t, draws[r, t]] = 1
    return np.cumsum(onehot, 1)                              # R x MMAX x k counts after t+1 draws

def top2(C):
    s = np.sort(C, -1)
    return s[..., -1], s[..., -2] if C.shape[-1] > 1 else np.zeros_like(s[..., -1])

def decide(Ct):
    # leader with random tie-break; -1 if no votes
    m = Ct.max(-1, keepdims=True)
    tie = (Ct == m) & (m > 0)
    noise = rng.random(Ct.shape) * tie
    d = noise.argmax(-1); d[m[..., 0] == 0] = -1
    return d

def policy_eval(C, stop_t, corr, mode):
    d = decide(C[np.arange(R), stop_t-1])
    acc = np.where(d >= 0, corr[np.clip(d, 0, None)], False).mean()
    return (stop_t.mean(), acc, (d == mode).mean())

def first_true(mask, default):
    has = mask.any(1)
    return np.where(has, mask.argmax(1) + 1, default)

def main():
    rows = {}
    def add(name, res): rows.setdefault(name, []).append(res)
    ks = [1, 3, 5, 9, 17, 33, 65, 129, 256]
    n_t = np.arange(1, MMAX+1)
    shares = []
    for q in qs:
        pr, _, va = prep(q); v = np.append(np.sort(np.where(va, pr[:-1], 0))[::-1], 0.0)
        shares.append(v[0]/(v[0]+v[1]) if v[0]+v[1] > 0 else 0.5)
    shares = np.clip(np.array(shares), 0.5, 1-1e-4)
    for qi, q in enumerate(qs):
        probs, corr, valid = prep(q)
        sh = np.delete(shares, qi)   # LOO prior support; symmetric: leader share s or 1-s
        sup = np.concatenate([sh, 1-sh]); lw = np.log(sup); l1w = np.log1p(-sup)
        mode = int(np.argmax(np.where(valid, probs[:-1], -1)))
        C = runs(probs, valid)
        a, b = top2(C)                                       # R x MMAX
        for k in ks:
            add(f'fixed k={k}', policy_eval(C, np.full(R, k), corr, mode))
            # curtailed: stop when the leader cannot be caught (strict lead > remaining)
            add(f'curtailed k={k}', policy_eval(C, first_true((a - b > k - n_t[None]) & (n_t[None] <= k), k), corr, mode))
        # certified anytime: leader-vs-runner-up truncated Beta(1,1) mixture LR vs 1/2 at level alpha (approximate)
        for alpha in [0.2, 0.05]:
            Mup = betaln(a+1, b+1) + np.log(np.clip(1 - betainc(a+1, b+1, 0.5), 1e-300, None)) + np.log(2) + (a+b)*np.log(2)
            add(f'certified a={alpha}', policy_eval(C, first_true(Mup >= np.log(1/alpha), MMAX), corr, mode))
        # Adaptive-Consistency-style Bayesian stop: P(p_leader > p_runner | Beta(1+a,1+b)) >= conf
        for conf, cap in [(0.9, 64), (0.95, 64), (0.99, 64), (0.95, 256)]:
            pb = Beta.sf(0.5, a+1, b+1)
            add(f'bayes conf={conf} cap={cap}', policy_eval(C, first_true((pb >= conf) & (n_t[None] <= cap), cap), corr, mode))
        A = a[..., None]*lw + b[..., None]*l1w; A -= A.max(-1, keepdims=True); E = np.exp(A)
        peb = (E*(sup > 0.5)).sum(-1)/E.sum(-1)
        for conf in [0.9, 0.95, 0.99, 0.999]:
            add(f'EB conf={conf} cap=256', policy_eval(C, first_true(peb >= conf, MMAX), corr, mode))
        # streak rule: first m votes unanimous -> stop at m, else fixed k=33 (curtailed)
        for m in [3, 5]:
            st = np.where((a[:, m-1] == m), m, first_true((a - b > 33 - n_t[None]) & (n_t[None] <= 33), 33))
            add(f'streak m={m} else curt33', policy_eval(C, st, corr, mode))
        add('_oracle mode (k=inf)', (np.inf, float(corr[mode]), 1.0))
        add('_pass@any', (np.inf, float(corr.any()), np.nan))
    print(f'# {sys.argv[1]}  questions={len(qs)} R={R}')
    print(f'{"policy":32s} {"mean n":>8s} {"acc":>7s} {"=mode":>7s}')
    curt = np.array([[np.mean([r[0] for r in rows[f'curtailed k={k}']]), np.mean([r[1] for r in rows[f'curtailed k={k}']])] for k in ks])
    def curt_at(acc):
        if acc > curt[:,1].max(): return np.nan
        i = np.searchsorted(curt[:,1], acc); i = max(i, 1)
        (n0, a0), (n1, a1) = curt[i-1], curt[i]
        return n0 + (n1-n0)*(acc-a0)/max(a1-a0, 1e-12)
    print(f'{"policy":32s} {"mean n":>8s} {"acc":>7s} {"=mode":>7s}  curtailed-n@acc  ratio')
    for name, v in rows.items():
        v = np.array(v); n, acc = v[:,0].mean(), v[:,1].mean()
        cn = curt_at(acc) if np.isfinite(n) and not name.startswith(('fixed','curtailed')) else np.nan
        extra = f'  {cn:10.1f}  {cn/n:5.2f}x' if np.isfinite(cn) else ''
        print(f'{name:32s} {n:8.2f} {acc:7.4f} {np.nanmean(v[:,2]):7.4f}{extra}')
main()
