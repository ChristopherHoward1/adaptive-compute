"""Exp 5: empirical-Bayes stopping across a batch. Does a learned-prior DP recover the oracle?
Batch of m cases, p_i from the Laplace-logit prior around c. Pilot n0 draws each, NPMLE prior on a grid via EM,
then DP-optimal continuation (with fitted prior) from each case's pilot state. Compared to oracle DP (true prior)
and curtailed fixed-B, all on the same cases/streams."""
import sys, numpy as np
from scipy.special import expit, logit, gammaln
c, tau, Nmax, m, n0 = float(sys.argv[1]), float(sys.argv[2]), int(sys.argv[3]), int(sys.argv[4]), int(sys.argv[5])
rng = np.random.default_rng(int(sys.argv[6]) if len(sys.argv) > 6 else 1)
G = 301
zg = np.linspace(-8*tau, 8*tau, G); zg = zg[zg != 0]; pg = expit(logit(c) + zg)
w_true = np.exp(-np.abs(zg)/tau); w_true /= w_true.sum()
lp, l1p = np.log(pg), np.log1p(-pg)

def policy(w, L):
    """returns list of boolean 'continue' rows cont[n][S]"""
    above = pg > c; Vn = None; conts = [None]*(Nmax+1)
    for n in range(Nmax, -1, -1):
        S = np.arange(n+1)
        ll = S[:, None]*lp + (n-S)[:, None]*l1p + np.log(w + 1e-300)
        ll -= ll.max(1, keepdims=True); post = np.exp(ll); post /= post.sum(1, keepdims=True)
        Pa = post[:, above].sum(1); stopV = L*np.minimum(Pa, 1-Pa)
        if n == Nmax: V = stopV; cont = np.zeros(n+1, bool)
        else:
            q = post @ pg; contV = 1 + q*Vn[1:] + (1-q)*Vn[:-1]; cont = contV < stopV; V = np.where(cont, contV, stopV)
        conts[n] = (cont, Pa > 0.5); Vn = V
    return conts

def run(conts, X, start):
    ns, wrong = [], []
    for x, p in zip(X, ps):
        n = start; S = int(x[:n].sum())
        while n < Nmax and conts[n][0][S]:
            S += x[n]; n += 1
        ns.append(n); wrong.append(conts[n][1][S] != (p > c))
    return np.mean(ns), np.mean(wrong)

def npmle(Sp, iters=500):
    w = np.ones(len(pg))/len(pg)
    lik = np.exp(gammaln(n0+1)-gammaln(Sp+1)-gammaln(n0-Sp+1))[:, None] * pg**Sp[:, None] * (1-pg)**(n0-Sp[:, None])
    for _ in range(iters):
        post = lik*w; post /= post.sum(1, keepdims=True); w = post.mean(0)
    return w

ps = expit(logit(c) + rng.laplace(0, tau, m))
X = (rng.random((m, Nmax)) < ps[:, None]).astype(int)
w_eb = npmle(X[:, :n0].sum(1))
print(f'# c={c} tau={tau} Nmax={Nmax} batch m={m} pilot n0={n0}')
print(f'{"L":>7s} | {"oracle n":>9s} {"err":>6s} | {"EB n":>9s} {"err":>6s} | {"EB-vs-oracle cost":>8s}')
for L in [300, 1000, 3000, 10000]:
    o = run(policy(w_true, L), X, 0); e = run(policy(w_eb, L), X, n0)
    print(f'{L:7d} | {o[0]:9.1f} {o[1]:6.3f} | {e[0]:9.1f} {e[1]:6.3f} | {e[0]/o[0]:8.2f}x')
# curtailed fixed-B on the same streams
for B in [63, 127, 255, 511]:
    k = int(np.floor(c*B)); ns = []; wr = []
    for x, p in zip(X, ps):
        cs = np.cumsum(x[:B]); t = np.arange(1, B+1)
        stop = np.nonzero((cs >= k+1) | (t - cs >= B-k))[0]; ns.append(stop[0]+1 if len(stop) else B)
        wr.append((cs[-1] > k) != (p > c))
    print(f'curtailed B={B:4d}: n={np.mean(ns):7.1f} err={np.mean(wr):.3f}')
